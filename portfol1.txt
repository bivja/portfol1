    const btn = this.querySelector('button[type="submit"]');

    btn.textContent = 'Заявка принята ✓';
    btn.classList.add('bg-white', 'text-dark');
    btn.classList.remove('bg-neon');
    status.textContent = 'Мы свяжемся с вами в ближайшее время.';
    status.classList.remove('hidden');

    setTimeout(() => {
      this.reset();
      btn.textContent = 'Подобрать ПК';
      btn.classList.remove('bg-white', 'text-dark');
      btn.classList.add('bg-neon');
      status.classList.add('hidden');
    }, 4000);
  });
